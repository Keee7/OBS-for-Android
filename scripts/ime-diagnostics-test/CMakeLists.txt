cmake_minimum_required(VERSION 3.20)
project(ime_diagnostics_test LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(ime_diagnostics_test main.cpp)
target_compile_features(ime_diagnostics_test PRIVATE cxx_std_17)
target_compile_definitions(ime_diagnostics_test PRIVATE OBS_IME_DIAGNOSTICS_TEST)
target_link_libraries(ime_diagnostics_test PRIVATE Qt6::Widgets)
enable_testing()
add_test(NAME diagnostics_disabled COMMAND ime_diagnostics_test disabled)
set_tests_properties(diagnostics_disabled PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;OBS_ANDROID_IME_DIAGNOSTICS=0")
add_test(NAME diagnostics_enabled COMMAND ime_diagnostics_test enabled)
set_tests_properties(diagnostics_enabled PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;OBS_ANDROID_IME_DIAGNOSTICS=1")
