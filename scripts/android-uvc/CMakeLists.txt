cmake_minimum_required(VERSION 3.22)
project(obs_android_uvc C)
if(NOT DEFINED DEPS_SOURCE_ROOT OR NOT DEFINED ANDROID_DEPS_PREFIX)
  message(FATAL_ERROR "DEPS_SOURCE_ROOT and ANDROID_DEPS_PREFIX are required")
endif()
add_library(LibUSB::LibUSB STATIC IMPORTED GLOBAL)
set_target_properties(LibUSB::LibUSB PROPERTIES
  IMPORTED_LOCATION "${ANDROID_DEPS_PREFIX}/lib/libusb-1.0.a"
  INTERFACE_INCLUDE_DIRECTORIES "${ANDROID_DEPS_PREFIX}/include/libusb-1.0")
add_library(JPEG::JPEG STATIC IMPORTED GLOBAL)
set_target_properties(JPEG::JPEG PROPERTIES
  IMPORTED_LOCATION "${ANDROID_DEPS_PREFIX}/lib/libjpeg.a"
  INTERFACE_INCLUDE_DIRECTORIES "${ANDROID_DEPS_PREFIX}/include")
set(JPEG_FOUND TRUE)
set(CMAKE_BUILD_TARGET Static CACHE STRING "" FORCE)
set(BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS_SOURCE_ROOT}/libuvc-0.0.7" libuvc)
